# A shape may contain nothing but these characters, which is what makes it
# structurally incapable of carrying a corpus value. profiler/schema.json pins the
# same alphabet, and profiler/selfcheck.py re-checks it.
SHAPE_SEPARATORS = "-./:"
SHAPE_ALPHABET = "ALaD_" + SHAPE_SEPARATORS + "#"
SHAPE_MAX = 16


def shape_of(value: str, limit: int = SHAPE_MAX) -> str:
    """Structure without content: A=first uppercase, L=later uppercase, a=lowercase,
    D=digit, _=whitespace, -./: kept because they distinguish identifier formats,
    # everything else.

    'F1247' -> 'ADDDD' keeps the leading-letter/four-digit structure while discarding
    every digit, so shapes are safe to emit from the profiler. Capped at `limit`
    characters: a longer shape starts to fingerprint the value it came from."""
    out = []
    for i, ch in enumerate(value[:limit]):
        if ch.isdigit():
            out.append("D")
        elif ch.isupper():
            out.append("A" if i == 0 else "L")
        elif ch.islower():
            out.append("a")
        elif ch.isspace():
            out.append("_")
        elif ch in SHAPE_SEPARATORS:
            out.append(ch)
        else:
            out.append("#")
    return "".join(out)
```

# profiler/stats/customdata.py
"""Profile the customData column: structure only, never its values.

Round 1 measured this field's length and never looked inside it, despite it being
present in nearly every row at a median length of a few hundred characters. If the client records error
codes here, the identifier retrieval arm is well-founded after all - and that is
decidable from key names and value SHAPES without emitting a single value.

Nothing here emits a value. Key names are emitted only after clearing a presence floor
and the entity sweep; values contribute only counts, quantiles and shapes.
"""

from __future__ import annotations

import json
import re

_KV_PAIR = re.compile(r"^\s*([^=;\n]{1,64})=([^;\n]*)\s*$")
_KV_SPLIT = re.compile(r"[;\n]")
_NUMERIC = re.compile(r"^\d+([.,]\d+)?$")
_WORD = re.compile(r"[A-Za-zÄÖÜäöüß]{2,}")

# A key must be present in this share of ALL rows as well as clearing the absolute
# floor. The absolute floor is the main control; this guards small corpora.
PRESENCE_SHARE_FLOOR = 0.01


def _flatten(obj: dict, prefix: str = "") -> dict[str, str]:
    out: dict[str, str] = {}
    for key, value in obj.items():
        path = f"{prefix}{key}"
        if isinstance(value, dict):
            out.update(_flatten(value, f"{path}."))
        elif isinstance(value, list):
            out[path] = " ".join(str(v) for v in value)
        else:
            out[path] = "" if value is None else str(value)
    return out


def parse(value: str | None) -> dict[str, str] | None:
    """JSON object first, then key=value pairs. None when neither applies.

    Never raises: a malformed value is a count, not an exception. The offending text
    must not travel, so nothing about it is returned or logged here."""
    if not value or not value.strip():
        return None
    text = value.strip()
    if text.startswith("{"):
        try:
            loaded = json.loads(text)
        except (ValueError, RecursionError):
            return None
        if isinstance(loaded, dict):
            return _flatten(loaded)
        return None
    pairs: dict[str, str] = {}
    for chunk in _KV_SPLIT.split(text):
        m = _KV_PAIR.match(chunk)
        if m:
            pairs[m.group(1).strip()] = m.group(2).strip()
    return pairs or None


def detect_format(values: list[str | None]) -> str:
    """Which shape does the majority of non-empty values take?"""
    non_empty = [v for v in values if v and v.strip()]
    if not non_empty:
        return "unknown"
    braces = sum(1 for v in non_empty if v.strip().startswith("{"))
    parsed = sum(1 for v in non_empty if parse(v) is not None)
    if braces / len(non_empty) >= 0.5:
        return "json"
    if parsed / len(non_empty) >= 0.5:
        return "key_value"
    return "unknown"


def classify(values: list[str]) -> str:
    """Which kind of data does this key hold? Rules are evaluated in order and the
    first match wins, so a key cannot be two kinds."""
    if not values:
        return "empty"
    total = len(values)
    non_empty = [v for v in values if v and v.strip()]
    if len(non_empty) / total < 0.05:
        return "empty"
    if sum(1 for v in non_empty if _NUMERIC.match(v.strip())) / len(non_empty) >= 0.90:
        return "numeric"
    id_like = sum(1 for v in non_empty if len(v) <= 32 and any(c.isdigit() for c in v))
    if id_like / len(non_empty) >= 0.70:
        return "identifier_like"
    tokens = [len(_WORD.findall(v)) for v in non_empty]
    mean_tokens = sum(tokens) / len(non_empty)
    mean_chars = sum(len(v) for v in non_empty) / max(1, sum(tokens))
    if mean_tokens >= 5 and mean_chars <= 12:
        return "prose"
    return "short_text"
```

# Longer than this and a character-class shape starts to fingerprint the value, because
# it preserves word lengths and punctuation positions.
MAX_SHAPED_VALUE_CHARS = 64
MAX_SHAPES_PER_KEY = 10


def profile(values: list[str | None], min_frequency: int, total_rows: int) -> dict:
    """Structure of the customData column. Emits key names, kinds, fill rates, value
    length quantiles and value shapes - never a value."""
    parsed_rows = [parse(v) for v in values]
    parsed_ok = [r for r in parsed_rows if r is not None]
    present = len([v for v in values if v and v.strip()]) or 1

    per_key: dict[str, list[str]] = defaultdict(list)
    for row in parsed_ok:
        for key, value in row.items():
            per_key[key].append(value)

    presence_floor = max(min_frequency, int(total_rows * PRESENCE_SHARE_FLOOR))
    keys_out: list[dict] = []
    suppressed = 0

    for key, key_values in sorted(per_key.items(), key=lambda kv: -len(kv[1])):
        if len(key_values) < presence_floor or entities.looks_like_entity(key):
            suppressed += 1
            continue
        non_empty = [v for v in key_values if v and v.strip()]
        kind = classify(key_values)
        shapeable = [v for v in non_empty if len(v) <= MAX_SHAPED_VALUE_CHARS]
        long_values = len(non_empty) - len(shapeable)
        shape_counts = Counter(shape_of(v) for v in shapeable) if kind != "prose" else Counter()
        kept = [
            {"shape": s, "count": c}
            for s, c in shape_counts.most_common()
            if c >= min_frequency
        ][:MAX_SHAPES_PER_KEY]
        keys_out.append(
            {
                "key": key,
                "present_count": len(key_values),
                "fill_rate": len(non_empty) / len(key_values),
                "kind": kind,
                "value_char_len_quantiles": lengths.quantiles([len(v) for v in non_empty]),
                "value_shapes": kept,
                "shapes_suppressed": max(0, len(shape_counts) - len(kept)),
                "long_value_count": long_values,
            }
        )

    return {
        "format": detect_format(values),
        "parse_success_rate": len(parsed_ok) / present,
        "keys_per_row_quantiles": lengths.quantiles([len(r) for r in parsed_ok]),
        "distinct_key_count": len(per_key),
        "keys_suppressed": suppressed,
        "keys": keys_out,
    }
```

# profiler/stats/shapes.py
"""Discover identifier formats instead of assuming them.

recall.german.identifiers matches patterns we wrote in advance. Round 1 reported a very
small number of identifier occurrences across the whole corpus, which means 'only those that matched our guesses'.
This histograms the character-class shape of every digit-bearing token, so a format
nobody anticipated - MOT-4711, 07.1234.556 - surfaces on its own.

Shapes are content-free by construction (see recall.german.identifiers.shape_of), and
rare shapes are dropped because that is where unusual values sit.
"""

from __future__ import annotations

import re
from collections import Counter, defaultdict

from recall.german.identifiers import shape_of

# Deliberately broader than the identifier patterns: a token may start with a digit
# (07.1234.556) and may carry separators that vocab.tokenize would split on.
_CANDIDATE = re.compile(r"[A-Za-z0-9][A-Za-z0-9._/:\-]{1,31}")

DEFAULT_LIMIT = 50


def discover(
    texts_by_column: dict[str, list[str]], min_frequency: int, limit: int = DEFAULT_LIMIT
) -> dict:
    """Shape histogram over digit-bearing tokens, floored and capped.

    Returns the kept shapes plus how many frequent ones the cap dropped. Discovery is
    the section that answers 'what formats does this corpus actually use', so a silently
    truncated list would be read as the complete answer.
    """
    counts: Counter[str] = Counter()
    fields: dict[str, set[str]] = defaultdict(set)

    for column, texts in texts_by_column.items():
        for text in texts:
            for match in _CANDIDATE.finditer(text or ""):
                token = match.group(0)
                if not any(c.isdigit() for c in token):
                    continue
                shape = shape_of(token)
                counts[shape] += 1
                fields[shape].add(column)

    total = sum(counts.values()) or 1
    above_floor = [
        {
            "shape": shape,
            "count": count,

def _fill_by_month(rows: list[dict], column: str) -> dict[str, float]:
    """Non-empty rate per month. A flat line means a workflow reality; a step at a date
    means a system change, and recent data may be better than the corpus average."""
    seen: dict[str, list[int]] = defaultdict(list)

        if len(month) == 7:
            seen[month].append(1 if (row.get(column) or "").strip() else 0)
    return {m: sum(v) / len(v) for m, v in sorted(seen.items())}

MAX_CHARACTERISED = 20


def characterise(
    docs: list[str], min_frequency: int, is_entity, threshold: float = 0.7, num_perm: int = 128
) -> tuple[list[dict], int]:
    """Top tokens of the largest near-duplicate clusters, so a dominant group can be
    identified. Clusters below min_frequency members are not characterised: a
    three-ticket cluster's top tokens is a summary of three tickets."""
    lsh = MinHashLSH(threshold=threshold, num_perm=num_perm)
    hashes = {}
    for i, doc in enumerate(docs):
        h = _minhash(doc, num_perm)
        hashes[i] = h
        lsh.insert(str(i), h)

    seen: set[int] = set()
    groups: list[set[int]] = []
    for i in range(len(docs)):
        if i in seen:
            continue
        members = ({int(k) for k in lsh.query(hashes[i])} | {i}) - seen
        seen |= members
        groups.append(members)

    big = sorted((g for g in groups if len(g) >= min_frequency), key=len, reverse=True)
    total = len(docs) or 1
    out = []
    for group in big[:MAX_CHARACTERISED]:
        counts = vocab.token_counts([docs[i] for i in group])
        out.append(
            {
                "size": len(group),
                "share": len(group) / total,
                "top_tokens": [
                    t["token"]
                    for t in vocab.top_tokens(counts, min_frequency, 15)
                    if not is_entity(t["token"])
                ],
            }
        )
    return out, max(0, len(big) - MAX_CHARACTERISED)
```

```python
# profiler/blessing.py
"""Whole-artifact catch-all: no string leaves without an explicit reason to.

Every per-section check in selfcheck.py protects a path someone remembered to protect.
dimension_hints[].id was emitted verbatim, with neither a frequency floor nor an entity
sweep, precisely because nobody thought to add one. This walks the finished artifact and
rejects any string that is not accounted for, so a new section has to declare how its
strings are protected rather than being trusted by default.

What it does not do is re-check a path that has been explicitly exempted below. Those
paths carry their own floor and entity sweep in selfcheck, and the exemption is what
lets this run as a hard abort instead of a warning. So this catches the *unanticipated*
string - a new field, a new section - which is the shape the dimension_hints bug had.
Adding a path to either list below is a deliberate act that moves the protection
elsewhere; it is not a way to quiet a failure.
"""

from __future__ import annotations

import re

from recall.german.identifiers import SHAPE_ALPHABET, SHAPE_MAX

from profiler.columns import EXPECTED_COLUMNS

# Both the alphabet and the cap come from the constants shape_of actually uses. A
# transcribed 16 here would silently start rejecting real shapes the day SHAPE_MAX moves.
_SHAPE = re.compile(f"^[{re.escape(SHAPE_ALPHABET)}]{{1,{SHAPE_MAX}}}$")
_STRUCTURAL_KEY = re.compile(r"^p\d+$|^max$|^\d{4}-\d{2}$|^\d+$")

# Strings we wrote ourselves. Blessed by PATH, never by value, so the same text
# appearing somewhere unexpected is still rejected.
LITERAL_PATHS = frozenset(
    {
        "schema_version",
        "source.encoding_detected",
        "source.delimiter",
        "source.sha256",
        "source.filename",
        "clusters.method",
        "model_designations.inferred_family_rule",
        "multi_fault_estimate.heuristic",
        "solutions.junk_taxonomy.pattern",
        "solutions.cardinality.detected_separator",
        "identifier_grammars.kind",
        "identifier_grammars.regex",
        "custom_data.format",
        "custom_data.keys.kind",
        "selfcheck.checks",
        "selfcheck.warnings",
    }
)

# Values at these paths cleared a frequency floor and the entity sweep when they were
# built, and selfcheck re-checks them section by section.
FLOORED_PATHS = frozenset(
    {
        "per_field.top_values.value",
        "dimension_hints.id",
        "custom_data.keys.key",
    }
)


def _declared_keys(schema: dict) -> set[str]:
    found: set[str] = set()

    def walk(node):
        if not isinstance(node, dict):
            return
        for key, value in node.get("properties", {}).items():
            found.add(key)
            walk(value)
        for nested in ("additionalProperties", "items"):
            walk(node.get(nested))
        for definition in node.get("$defs", {}).values():
            walk(definition)

    walk(schema)
    return found


def generic_path(path: str, columns: set[str]) -> str:
    """Collapse a concrete path to the one the exemption lists are written in.

    per_field and dimension_hints are dicts keyed by column name, so a real path looks
    like per_field.categoryId.top_values.0.value. Stripping only the list indices leaves
    the column in place, and the declared exemption per_field.top_values.value then never
    matches anything - it reads as deliberate protection while doing nothing at all. Two
    entries were dead exactly that way when this was checked against the round-1 artifact.

    A column name is collapsed only in the middle of a path. First segment, because
    'solutions' is both a column and a top-level section; last segment, because 'id' is
    also a column name and stripping it would collapse dimension_hints.id to
    dimension_hints.
    """
    parts = path.split(".")
    last = len(parts) - 1
    return ".".join(
        part
        for i, part in enumerate(parts)
        if not part.isdigit() and (i == 0 or i == last or part not in columns)
    )


def _collect_floored(profile: dict) -> set[str]:
    """Every string that some floored, entity-swept list already vouches for."""
    blessed: set[str] = set()
    for key in ("top_tokens", "top_bigrams", "hunspell_unrecognized_top"):
        for entry in profile.get("vocabulary", {}).get(key, []):
            blessed.add(entry["token"])
    for stats in profile.get("per_field", {}).values():
        for entry in stats.get("top_values", []):
            blessed.add(entry["value"])
    for hints in profile.get("dimension_hints", {}).values():
        for hint in hints:
            blessed.add(str(hint.get("id")))
            blessed.update(hint.get("top_tokens", []))
    for cluster in profile.get("clusters", {}).get("largest", []):
        blessed.update(cluster.get("top_tokens", []))
    for entry in profile.get("custom_data", {}).get("keys", []):
        blessed.add(entry["key"])
    return blessed


def unblessed(profile: dict, schema: dict | None = None) -> list[tuple[str, str]]:
    """Return (schema path, offending string) for everything unaccounted for."""
    if schema is None:
        import json
        import pathlib

        schema = json.loads(
            (pathlib.Path(__file__).parent / "schema.json").read_text(encoding="utf-8")
        )

    declared = _declared_keys(schema)
    columns = set(EXPECTED_COLUMNS) | set(profile.get("source", {}).get("columns_found", []))
    floored = _collect_floored(profile)
    custom_keys = {e["key"] for e in profile.get("custom_data", {}).get("keys", [])}
    found: list[tuple[str, str]] = []

    def walk(node, path: str) -> None:
        if isinstance(node, dict):
            for key, value in node.items():
                if not (
                    key in declared
                    or key in columns
                    or _STRUCTURAL_KEY.match(key)
                    or key in custom_keys
                ):
                    found.append((path or "<root>", key))
                walk(value, f"{path}.{key}" if path else key)
        elif isinstance(node, list):
            for index, value in enumerate(node):
                walk(value, f"{path}.{index}")
        elif isinstance(node, str):
            base = generic_path(path, columns)
            if base in LITERAL_PATHS or base in FLOORED_PATHS:
                return
            if node in columns or _SHAPE.match(node) or node in floored:
                return
            found.append((base, node))

    walk(profile, "")
    return found
```
